cmake_minimum_required(VERSION 3.16)
project(wxraw LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Gui)
add_executable(wxraw MACOSX_BUNDLE main.cpp)
set_target_properties(wxraw PROPERTIES OUTPUT_NAME "WX Raw" MACOSX_BUNDLE_GUI_IDENTIFIER io.github.loomhand.wxfocus.raw)
target_link_libraries(wxraw PRIVATE Qt6::Gui)
